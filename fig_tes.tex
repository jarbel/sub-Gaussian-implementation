  \begin{axis}[

    xtick={-1,0,1},
    xticklabels={$-a$,$0$,$a$},
    ytick=\empty,             % aucune graduation en y
    grid=none,
    width=11cm, height=6.5cm,
    clip=false
  ]
    % PMF
    \addplot+[ycomb, thick, mark=*, mark options={solid}] coordinates
      {(-1,\p) (0,{1-2*\p}) (1,\p)};

    % Étiquettes au-dessus des bâtons
    \node[above] at (axis cs:-1,\p) {$p$};
    \node[above] at (axis cs:0,{1-2*\p}) {$1-2p$};
    \node[above] at (axis cs:1,\p) {$p$};
  \end{axis}
